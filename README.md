        /* -----------------------------------------------------------------
           4. SLIDER 2 BANNER TỰ ĐỘNG TRƯỢT (Đồng bộ cả Máy tính & Điện thoại)
           ----------------------------------------------------------------- */
        let currentSlideIndex = 0;
        let bannerInterval = null;

        function updateBannerView() {
            for (let i = 0; i < 2; i++) {
                const dSlide = document.getElementById('desktopSlide' + i);
                const mSlide = document.getElementById('mobileSlide' + i);
                if (dSlide) {
                    dSlide.style.opacity = (i === currentSlideIndex) ? '1' : '0';
                    dSlide.style.zIndex = (i === currentSlideIndex) ? '10' : '0';
                }
                if (mSlide) {
                    mSlide.style.opacity = (i === currentSlideIndex) ? '1' : '0';
                    mSlide.style.zIndex = (i === currentSlideIndex) ? '10' : '0';
                }
            }

            const dDots = document.querySelectorAll('.banner-dot-desktop');
            dDots.forEach((dot, idx) => {
                dot.className = (idx === currentSlideIndex)
                    ? 'banner-dot-desktop w-6 h-2 rounded-full bg-amber-400 transition-all'
                    : 'banner-dot-desktop w-2.5 h-2 rounded-full bg-white/60 transition-all';
            });

            const mDots = document.querySelectorAll('.banner-dot-mobile');
            mDots.forEach((dot, idx) => {
                dot.className = (idx === currentSlideIndex)
                    ? 'banner-dot-mobile w-4 h-1.5 rounded-full bg-amber-400 transition-all'
                    : 'banner-dot-mobile w-1.5 h-1.5 rounded-full bg-white/60 transition-all';
            });
        }

        function changeBannerSlide(step) {
            currentSlideIndex = (currentSlideIndex + step + 2) % 2;
            updateBannerView();
            resetBannerTimer();
        }

        function setBannerSlide(index) {
            currentSlideIndex = index;
            updateBannerView();
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                currentSlideIndex = (currentSlideIndex + 1) % 2;
                updateBannerView();
            }, 5000);
        }
